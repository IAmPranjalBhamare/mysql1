CREATE PROCEDURE CALCULATEDELAY(IN P_SHIPMENTID INT)
BEGIN
	DECLARE V_EXPDATE DATE;
	DECLARE V_ACTUALDATE DATE;
	DECLARE V_DELAY INT ;
	DECLARE V_COUNT INT DEFAULT 0;
	
	DECLARE EXIT HANDLER FOR SQLEXCEPTION
	BEGIN 
		RESIGNAL;
	END;

	SELECT COUNT(*) INTO V_COUNT 
	FROM SHIPMENT 
	WHERE SHIPMENTID = P_SHIPMENTID;

	IF V_COUNT = 0 THEN
		SIGNAL SQLSTATE '45000'
		SET MESSAGE_TEXT ='SHIPMENT ID NOT FOUND';	
	END IF;

	SELECT COUNT(*) INTO V_COUNT 
	FROM DELIVERY 
	WHERE SHIPMENTID = P_SHIPMENTID;

	IF V_COUNT = 0 THEN 
		SIGNAL SQLSTATE '45000'
		SET MESSAGE_TEXT='DELIVERY DETAILS MISSING';
	END IF;

	SELECT EXPECTEDDELIVERYDATE INTO V_EXPDATE
	FROM SHIPMENT 
	WHERE SHIPMENTID=P_SHIPMENTID;

	SELECT ACTUALDELIVERYDATE INTO V_ACTUALDATE
	FROM DELIVERY 
	WHERE SHIPMENTID=P_SHIPMENTID;

	IF V_ACTUALDATE IS NULL THEN 
		SIGNAL SQLSTATE '45000'
		SET MESSAGE_TEXT='DELIVERY DATE NOT FOUND';
	END IF;

	SET V_DELAY = GREATEST(0,DATEDIFF(V_ACTUALDATE,V_EXPDATE));

	UPDATE DELIVERY
	SET DELAYDAYS = V_DELAY
	WHERE SHIPMENTID = P_SHIPMENTID;

END //

DELIMITER ;

DELIMITER // 

CREATE PROCEDURE CALCULATEPENALTY ( IN P_SHIPMENTID INT)
BEGIN 

	DECLARE V_PENALTY INT DEFAULT 0;
	DECLARE V_DELAY INT;
	DECLARE I INT DEFAULT 1;
	

	CALL CALCULATEDELAY(P_SHIPMENTID);

	SELECT DELAYDAYS INTO V_DELAY
	FROM DELIVERY 
	WHERE SHIPMENTID = P_SHIPMENTID;

	IF V_DELAY > 0 THEN 
		WHILE I <= V_DELAY DO
			SET V_PENALTY = V_PENALTY + 10;
			SET I = I+ 1;
		END WHILE ;
	END IF ;


	UPDATE DELIVERY 
	SET PENALTYAMOUNT = V_PENALTY
	WHERE SHIPMENTID= P_SHIPMENTID;

END //

DELIMITER ;

DELIMITER //
CREATE PROCEDURE PROJECTBATTERYHEALTH(IN P_VEHICLE INT)
BEGIN
  DECLARE DONE INT DEFAULT 0;
  DECLARE BID INT;
  DECLARE CAP DECIMAL(8,2);
  DECLARE CURRENT_CAP DECIMAL(8,2);
  DECLARE HEALTH DECIMAL(5,2);
  DECLARE CUR CURSOR FOR
    SELECT B.BATTERYID,B.CAPACITYKWH,T.CURRENTCAPACITY
    FROM BATTERY B JOIN BATTERY_TEST T ON B.BATTERYID=T.BATTERYID
    WHERE B.VEHICLEID=P_VEHICLE;
  DECLARE CONTINUE HANDLER FOR NOT FOUND SET DONE=1;

  IF NOT EXISTS(SELECT 1 FROM VEHICLE WHERE VEHICLEID=P_VEHICLE) THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT='INVALID VEHICLE ID';
  END IF;
  OPEN CUR;
  HEALTH_LOOP: LOOP
    FETCH CUR INTO BID,CAP,CURRENT_CAP;
    IF DONE=1 THEN LEAVE HEALTH_LOOP; END IF;
    IF CAP IS NULL OR CURRENT_CAP IS NULL OR CAP<=0 OR CURRENT_CAP<0 THEN
      SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT='INVALID OR MISSING BATTERY READING';
    END IF;
    SET HEALTH=(CURRENT_CAP/CAP)*100;
    UPDATE BATTERY_TEST SET SOH=HEALTH,
      BATTERYSTATUS=IF(HEALTH>=80,'HEALTHY',IF(HEALTH>=60,'MODERATE','POOR'))
      WHERE BATTERYID=BID AND CURRENTCAPACITY=CURRENT_CAP;
  END LOOP;
  CLOSE CUR;
END//

CREATE FUNCTION PRORATEDCHARGE(P_OLD INT,P_NEW INT,P_DAYS INT)
RETURNS DECIMAL(10,2) DETERMINISTIC READS SQL DATA
BEGIN
  DECLARE OLD_PRICE DECIMAL(8,2);
  DECLARE NEW_PRICE DECIMAL(8,2);
  SELECT MONTHLYPRICE INTO OLD_PRICE FROM SUBSCRIPTION_PLAN WHERE PLANID=P_OLD;
  SELECT MONTHLYPRICE INTO NEW_PRICE FROM SUBSCRIPTION_PLAN WHERE PLANID=P_NEW;
  RETURN ((NEW_PRICE-OLD_PRICE)/30)*P_DAYS;
END//

CREATE PROCEDURE UPGRADEPLAN(IN P_USER INT, IN P_PLAN INT, OUT P_CHARGE DECIMAL(10,2))
BEGIN
  DECLARE OLD_PLAN INT;
  DECLARE DAYS_LEFT INT;
  SELECT PLANID,DATEDIFF(ENDDATE,CURDATE()) INTO OLD_PLAN,DAYS_LEFT
  FROM USER_SUBSCRIPTION WHERE USERID=P_USER;
  IF OLD_PLAN IS NULL OR NOT EXISTS(SELECT 1 FROM SUBSCRIPTION_PLAN WHERE PLANID=P_PLAN) THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT='INVALID USER SUBSCRIPTION OR PLAN';
  END IF;
  SET P_CHARGE=PRORATEDCHARGE(OLD_PLAN,P_PLAN,GREATEST(DAYS_LEFT,0));
  UPDATE USER_SUBSCRIPTION SET PLANID=P_PLAN WHERE USERID=P_USER;
END//

CREATE PROCEDURE PROCESSREVIEWS()
BEGIN
    DECLARE DONE INT DEFAULT 0;
    DECLARE SID,PID INT;
    DECLARE REVIEWER VARCHAR(50);
    DECLARE ST VARCHAR(20);

    DECLARE CUR CURSOR FOR
    SELECT STAGEID,PAPERID,REVIEWERNAME,REVIEWSTATUS
    FROM REVIEW_STAGE;

    DECLARE CONTINUE HANDLER FOR NOT FOUND SET DONE=1;

    OPEN CUR;

    REVIEW_LOOP: LOOP
        FETCH CUR INTO SID,PID,REVIEWER,ST;

        IF DONE=1 THEN
            LEAVE REVIEW_LOOP;
        END IF;

        IF ST IN ('PENDING','ACCEPTED','REJECTED')
           AND NOT EXISTS(
               SELECT 1 FROM REVIEW_MASTER WHERE STAGEID=SID
           ) THEN
            INSERT INTO REVIEW_MASTER(STAGEID,PAPERID,REVIEWERNAME,REVIEWSTATUS)
            VALUES(SID,PID,REVIEWER,ST);
        END IF;
    END LOOP;

    CLOSE CUR;
END//

DELIMITER ;

CREATE PROCEDURE PROCESSCALIBRATION()
BEGIN
  DECLARE DONE INT DEFAULT 0;
  DECLARE SID,EID INT;
  DECLARE CDATE,DDATE DATE;
  DECLARE RES VARCHAR(30);
  DECLARE CUR CURSOR FOR
    SELECT STAGEID,EQUIPMENTID,CALIBRATIONDATE,NEXTDUEDATE,RESULT FROM CALIBRATION_STAGE;
  DECLARE CONTINUE HANDLER FOR NOT FOUND SET DONE=1;
  OPEN CUR;
  CAL_LOOP: LOOP
    FETCH CUR INTO SID,EID,CDATE,DDATE,RES;
    IF DONE=1 THEN LEAVE CAL_LOOP; END IF;
    IF CDATE IS NOT NULL AND DDATE IS NOT NULL AND DDATE>CDATE
       AND NOT EXISTS(SELECT 1 FROM CALIBRATION_MASTER WHERE STAGEID=SID) THEN
      INSERT INTO CALIBRATION_MASTER(STAGEID,EQUIPMENTID,CALIBRATIONDATE,NEXTDUEDATE,RESULT)
      VALUES(SID,EID,CDATE,DDATE,RES);
    END IF;
  END LOOP;
  CLOSE CUR;
END//

);
CREATE TABLE SYSTEM_INFO(
  SYSTEMID INT PRIMARY KEY, SYSTEMNAME VARCHAR(50) NOT NULL
);
CREATE TABLE ACCESS_LOG(
  LOGID INT PRIMARY KEY, USERID INT, SYSTEMID INT,
  LOGINTIME DATETIME NOT NULL, LOGINSTATUS VARCHAR(20) NOT NULL,
  FOREIGN KEY(USERID) REFERENCES USER_ACCOUNT(USERID),
  FOREIGN KEY(SYSTEMID) REFERENCES SYSTEM_INFO(SYSTEMID),
  CHECK(LOGINSTATUS IN ('SUCCESS','FAILED'))
);
CREATE TABLE THREAT_ALERT(
  THREATID INT PRIMARY KEY, LOGID INT, THREATSTATUS VARCHAR(20),
  THREATTYPE VARCHAR(50), FOREIGN KEY(LOGID) REFERENCES ACCESS_LOG(LOGID)
);
CREATE TABLE ML_PREDICTION(
  PREDICTIONID INT PRIMARY KEY, LOGID INT, PREDICTION VARCHAR(50),
  CONFIDENCE DECIMAL(5,2), FOREIGN KEY(LOGID) REFERENCES ACCESS_LOG(LOGID)
);
